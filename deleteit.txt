fetch('http://localhost:3000/users', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ username: 'Ivan', email: 'ivan@mail.com' })
})
.then(res => res.json())
.then(data => console.log('Создан:', data));


fetch('http://localhost:3000/users/2', {
  method: 'PUT',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ username: 'UpdatedName', email: 'new@mail.com' })
})
.then(res => res.json())
.then(data => console.log('Обновлен:', data));


fetch('http://localhost:3000/users/2', {
  method: 'DELETE'
})
.then(res => res.json())
.then(data => console.log('Результат удаления:', data));

const requireAuthorization = (req, res, next) => {
    if (!req.headers.authorization) {
        return res.status(401).json({ message: 'Требуется заголовок Authorization' });
    }
    next();
};

app.get('/admin', requireAuthorization, (req, res) => {
    res.json({ message: 'Добро пожаловать в админ-панель' });
});
